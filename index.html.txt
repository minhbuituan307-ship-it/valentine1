        function unlock() {
            const pinInput = document.getElementById('pin').value;
            if (pinInput === "1803") {
                document.getElementById('lock-screen').style.transform = "translateY(-100%)";
                document.getElementById('main-content').style.display = "flex";
                if (player && player.playVideo) { 
                    player.unMute(); 
                    player.setVolume(100); 
                    player.playVideo(); 
                }
                calculateDays("2025-07-07"); 
                setInterval(createHeart, 300);
            } else { alert("Try again, babe! ❤️"); }
        }

        function calculateDays(startDateStr) {
            const start = new Date(startDateStr);
            const now = new Date();
            const diff = Math.floor((now - start) / (1000 * 60 * 60 * 24));
            document.getElementById('days-number').innerText = diff;
        }

        function moveNo() {
            const btn = document.getElementById('no-btn');
            const container = document.querySelector('.btn-container');
            const newX = Math.random() * (container.offsetWidth - btn.offsetWidth);
            const newY = Math.random() * (container.offsetHeight - btn.offsetHeight);
            btn.style.left = newX + 'px';
            btn.style.top = newY + 'px';
        }

        function createHeart() {
            const heart = document.createElement('div');
            heart.classList.add('heart-float');
            heart.innerHTML = '❤️';
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.animationDuration = (Math.random() * 2 + 3) + 's';
            document.body.appendChild(heart);
            setTimeout(() => heart.remove(), 4000);
        }

        function sayYes() {
            document.getElementById('main-content').style.display = 'none';
            document.getElementById('success-screen').style.display = 'block';
            document.body.style.overflowY = "auto";
            for(let i=0; i<30; i++) { setTimeout(createHeart, i*100); }
        }

        function showLetter() {
            document.getElementById('letter-box').style.display = 'block';

            document.getElementById('footer-msg').style.display = 'block';
            document.getElementById('letter-box').scrollIntoView({ behavior: 'smooth' });
        }
